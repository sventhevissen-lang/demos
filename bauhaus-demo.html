    const defaults = [
      { name: "Jamal Musiala", val: 140 },
      { name: "Florian Wirtz", val: 140 },
      { name: "Lamine Yamal", val: 180 }
    ];
    let players = JSON.parse(localStorage.getItem("players_v1")) || defaults;

    function render() {
      localStorage.setItem("players_v1", JSON.stringify(players));
      document.getElementById("list").innerHTML = players.map((p, i) => `
        <div class="card">
          <div>
            <div class="name">${p.name}</div>
            <div class="val">Marktwert: <span>${p.val} Mio. €</span></div>
          </div>
          <div class="btns">
            <button class="plus" onclick="change(${i}, 5)">+5 Mio.</button>
            <button class="minus" onclick="change(${i}, -5)">-5 Mio.</button>
          </div>
        </div>
      `).join("");
    }
    function change(i, d) { players[i].val = Math.max(0, players[i].val + d); render(); }
    function addPlayer() {
      const n = document.getElementById("p-name").value.trim();
      const v = parseFloat(document.getElementById("p-val").value);
      if (!n || isNaN(v)) return;
      players.push({ name: n, val: v });
      document.getElementById("p-name").value = "";
      document.getElementById("p-val").value = "";
      render();
    }
    render();
  </script>
</body>
